% Chapter heading: "<Roman>.  <TITLE>", centered, 14 pt bold
\renewcommand{\thechapter}{\Roman{chapter}}
\titleformat{\chapter}[block]
  {\centering\normalfont\bfseries\fontsize{14}{16}\selectfont}
  {\thechapter.}
  {0.5em}
  {\fontsize{14}{16}\selectfont}
% 1 cm titlespacing + 2.5 cm top margin = 3.5 cm effective top margin on chapter pages
\titlespacing*{\chapter}{0pt}{1cm}{4\baselineskip}

% Sub-headings: 12 pt bold, number indented 1.25 cm
\titleformat{\section}{\normalfont\bfseries\fontsize{12}{14}\selectfont}{\thesection}{0.5em}{}
\titlespacing*{\section}{1.25cm}{2\baselineskip}{1\baselineskip}
\titleformat{\subsection}{\normalfont\bfseries\fontsize{12}{14}\selectfont}{\thesubsection}{0.5em}{}
\titlespacing*{\subsection}{1.25cm}{1\baselineskip}{1\baselineskip}
